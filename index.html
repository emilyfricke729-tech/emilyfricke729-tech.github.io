// Story-Validator: prüft alle Pfade
def validate_graph(nodes):
    visited = set()
    dead_ends = []

    for node in nodes:
        if not node.choices:
            dead_ends.append(node.id)
        visited.add(node.id)

    return {
        "valid": len(dead_ends) == 0,
        "dead_ends": dead_ends
    }

Der Thread läuft parallel zur Benutzereingabe und zeigt in Echtzeit den Countdown an — ohne die Hauptlogik zu blockieren.

Java Thread Daemon OOP 
// Asynchroner Timer-Thread für Echtzeit-Countdown
Thread timerThread = new Thread(() -> {
    while (true) {
        long elapsed   = (System.currentTimeMillis()
                          - startTime) / 1000;
        long remaining = timeLimitSec - elapsed;

        if (remaining < 0) break;

        System.out.print(
            "\r⏳ Zeit übrig: " + remaining + "s   "
        );
        try {
            Thread.sleep(500);
        } catch (InterruptedException e) {}
    }
});

// Automatisch beenden wenn Hauptthread endet
timerThread.setDaemon(true);
timerThread.start();
